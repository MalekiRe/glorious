run(value: Result<u8, Error>) -> u8 {
  handle perform Read<u8>::read(value) {
    Read<u8>::read(Result::Ok(byte)) with k => resume(k, move byte),
    Read<u8>::read(Result::Err(_)) with k => cancel(k)
  }
}
